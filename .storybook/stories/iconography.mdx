export const handleCopy = (name, ev) => {
  const cellEl = ev.currentTarget;
  const tip = cellEl.querySelector('[data-role="copied-tip"]');
  const reveal = () => {
    if (!tip) return;
    tip.style.opacity = '1';
    if (cellEl._copiedTimer) clearTimeout(cellEl._copiedTimer);
    cellEl._copiedTimer = setTimeout(() => {
      tip.style.opacity = '0';
    }, 1200);
  };
  if (navigator.clipboard?.writeText) {
    navigator.clipboard
      .writeText(name)
      .then(reveal)
      .catch(() => {
        // Fallback for environments where async clipboard is blocked.
        const ta = document.createElement('textarea');
        ta.value = name;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        try {
          document.execCommand('copy');
        } finally {
          document.body.removeChild(ta);
        }
        reveal();
      });
  }
};

# Iconography

These icons are designed to enhance visual communication, improve user experience, and maintain consistency across different interfaces and platforms. Click any icon to copy its name.

Every icon is one drawing per style, scaled through the `size` prop (16 / 20 / 24 / 32).
