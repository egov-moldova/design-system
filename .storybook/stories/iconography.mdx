export const handleCopy = (name, ev) => {
  const cellEl = ev.currentTarget;
  const tip = cellEl.querySelector('[data-role="copied-tip"]');
  const reveal = () => {
    if (!tip) return;
    tip.style.opacity = '1';
    if (cellEl._copiedTimer) clearTimeout(cellEl._copiedTimer);
    cellEl._copiedTimer = setTimeout(() => {
      tip.style.opacity = '0';
    }, 1200);
  };
  if (navigator.clipboard?.writeText) {
    navigator.clipboard
      .writeText(name)
      .then(reveal)
      .catch(() => {
        // Fallback for environments where async clipboard is blocked.
        const ta = document.createElement('textarea');
        ta.value = name;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        try {
          document.execCommand('copy');
        } finally {
          document.body.removeChild(ta);
        }
        reveal();
      });
  }
};

export const SizeSection = ({ size, names }) =>
  names.length ? (
    <section>
      <div style={headerWrap}>
        <p style={headerTitle}>Size</p>
        <span style={headerBadge}>{size}px</span>
      </div>
      <div style={grid}>
        {names.map(name => (
          <div
            key={name}
            style={cell}
            title={`Click to copy "${name}"`}
            role="button"
            tabIndex={0}
            onClick={ev => handleCopy(name, ev)}
            onKeyDown={ev => {
              if (ev.key === 'Enter' || ev.key === ' ') {
                ev.preventDefault();
                handleCopy(name, ev);
              }
            }}
          >
            <div style={iconWell}>
              <cor-icon name={name} size={size}></cor-icon>
            </div>
            <code style={label}>{name}</code>
            <span data-role="copied-tip" style={copiedToast}>
              Copied
            </span>
          </div>
        ))}
      </div>
    </section>
  ) : null;
